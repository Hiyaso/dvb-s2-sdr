cmake_minimum_required(VERSION 3.10)
project(dvbs2_b210 CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(UHD 4.0 REQUIRED)
find_package(Boost REQUIRED COMPONENTS program_options)
find_package(Threads REQUIRED)
add_executable(dvbs2_b210 dvbs2_b210.cpp)
target_include_directories(dvbs2_b210 PRIVATE ${UHD_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS})
target_link_libraries(dvbs2_b210 ${UHD_LIBRARIES} Boost::program_options Threads::Threads)
install(TARGETS dvbs2_b210 DESTINATION bin)
